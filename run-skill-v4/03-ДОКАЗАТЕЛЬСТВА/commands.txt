import math
def wilson(tp, n, z=1.96):
    if n == 0: return 0.0
    p = tp/n; den = 1+z*z/n; centre = p+z*z/(2*n)
    margin = z*math.sqrt((p*(1-p)+z*z/(4*n))/n)
    return max(0.0, (centre-margin)/den)

# Размеченные находки: 3 репозитория сессии + 2 мутанта (в области действия)
data = {
    "no_float_for_money":     (0, 2),   # 2 находки: комментарий с f64 и метрика задержки — не деньги
    "thiserror_for_errors":   (1, 1),
    "tech_radar_hold":        (1, 1),
    "tech_radar_renamed":     (1, 1),
    "authorize_idempotent":   (1, 1),
    "idempotency_contract":   (1, 1),
}
print(f"{'правило':<22}{'tp':>3}{'n':>3}{'точность':>10}{'нижняя':>9}  решение")
for k,(tp,n) in sorted(data.items(), key=lambda x: wilson(*x[1])):
    p = tp/n; low = wilson(tp,n)
    verdict = "показывать" if low >= 0.9 else ("сузить правило" if p < 0.9 else "разметить больше")
    print(f"{k:<22}{tp:>3}{n:>3}{p:>10.2f}{low:>9.2f}  {verdict}")
tot_tp = sum(v[0] for v in data.values()); tot_n = sum(v[1] for v in data.values())
print(f"\nитого: {tot_tp}/{tot_n} = {tot_tp/tot_n:.2f}, нижняя граница {wilson(tot_tp,tot_n):.2f} (порог 0.90)")
EOF
